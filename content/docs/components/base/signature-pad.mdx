## Usage

```tsx showLineNumbers
import {
  SignaturePad,
  SignaturePadCanvas,
  SignaturePadClear,
  SignaturePadSave,
  SignaturePadUndo,
  SignaturePadAction,
  SignaturePadActions,
} from "@/components/kromm-ui/signature-pad";
```

```tsx showLineNumbers
<SignaturePad>
  <SignaturePadCanvas />
  <SignaturePadActions>
    <SignaturePadAction>
      <SignaturePadUndo />
      <SignaturePadClear />
      <SignaturePadSave />
    </SignaturePadAction>
  </SignaturePadActions>
</SignaturePad>
```

## Composition

Use the following composition to build a `SignaturePad`:

```text
SignaturePad
├── SignaturePadCanvas
└── SignaturePadActions
    └── SignaturePadAction
        ├── SignaturePadUndo
        ├── SignaturePadClear
        └── SignaturePadSave
```
